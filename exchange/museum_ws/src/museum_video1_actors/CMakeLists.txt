cmake_minimum_required(VERSION 3.8)
project(museum_video1_actors)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(social_nav_msgs REQUIRED)
add_library(video1_actor SHARED src/actor_plugin.cpp)
target_include_directories(video1_actor PRIVATE include ${GAZEBO_INCLUDE_DIRS})
target_link_directories(video1_actor PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(video1_actor ${GAZEBO_LIBRARIES})
ament_target_dependencies(video1_actor gazebo_ros rclcpp social_nav_msgs)
install(TARGETS video1_actor LIBRARY DESTINATION lib)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/people_bridge.py scripts/record_demo.py scripts/social_markers.py
  DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/demo_math.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(test_actor_motion test/test_actor_motion.cpp)
  target_include_directories(test_actor_motion PRIVATE include)
  add_test(NAME video1_actor_motion COMMAND test_actor_motion)
  add_test(NAME video1_demo_tools COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_demo_tools.py)
endif()
ament_package()
